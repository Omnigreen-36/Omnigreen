Serial Bluetooth Terminal (compatible android)

Commande	Action
Z	       Avancer
S	       Reculer
Q	       Gauche
D	       Droite
A	       Rotation gauche
E	       Rotation droite
R	       Stop
W	       Lames ON
X	       Lames OFF

// =====================
// VITESSE
// =====================
int vitesse = 180;   // 0 à 255
int vitesseLames = 200;

// =====================
// SETUP
// =====================
void setup() {

  Serial.begin(9600);

  // Moteurs roues
  pinMode(4, OUTPUT); // IN1
  pinMode(5, OUTPUT); // IN2
  pinMode(6, OUTPUT); // IN3
  pinMode(7, OUTPUT); // IN4
  pinMode(3, OUTPUT); // ENA
  pinMode(9, OUTPUT); // ENB

  // Moteurs lames
  pinMode(8, OUTPUT);   // IN1
  pinMode(10, OUTPUT);  // IN2
  pinMode(11, OUTPUT);  // IN3
  pinMode(12, OUTPUT);  // IN4
  pinMode(2, OUTPUT);   // ENA
  pinMode(13, OUTPUT);  // ENB

  stopMoteurs();
}

// =====================
// LOOP
// =====================
void loop() {

  if (Serial.available()) {

    char c = Serial.read();

    switch(c) {
      case 'Z': avancer(); break;
      case 'S': reculer(); break;
      case 'Q': gauche(); break;
      case 'D': droite(); break;
      case 'A': rotationGauche(); break;
      case 'E': rotationDroite(); break;
      case 'R': stopMoteurs(); break;
      case 'W': lamesON(); break;
      case 'X': lamesOFF(); break;
    }
  }
}

// =====================
// MOUVEMENTS
// =====================

void avancer() {
  moteurGauche(HIGH, LOW);
  moteurDroit(HIGH, LOW);
}

void reculer() {
  moteurGauche(LOW, HIGH);
  moteurDroit(LOW, HIGH);
}

void gauche() {
  moteurGauche(LOW, LOW);     // arrêt gauche
  moteurDroit(HIGH, LOW);    // avance droite
}

void droite() {
  moteurGauche(HIGH, LOW);
  moteurDroit(LOW, LOW);
}

void rotationGauche() {
  moteurGauche(LOW, HIGH);
  moteurDroit(HIGH, LOW);
}

void rotationDroite() {
  moteurGauche(HIGH, LOW);
  moteurDroit(LOW, HIGH);
}

void stopMoteurs() {

  // Couper PWM
  analogWrite(3, 0);
  analogWrite(9, 0);

  // Couper directions (IMPORTANT)
  digitalWrite(4, LOW);
  digitalWrite(5, LOW);
  digitalWrite(6, LOW);
  digitalWrite(7, LOW);
}

// =====================
// CONTROLE MOTEURS
// =====================

void moteurGauche(int sens1, int sens2) {
  digitalWrite(4, sens1);
  digitalWrite(5, sens2);
  analogWrite(3, vitesse);
}

void moteurDroit(int sens1, int sens2) {
  digitalWrite(6, sens1);
  digitalWrite(7, sens2);
  analogWrite(9, vitesse);
}

// =====================
// LAMES
// =====================

void lamesON() {

  // Lame 1
  digitalWrite(8, HIGH);
  digitalWrite(10, LOW);
  digitalWrite(2, HIGH);   // ON simple

  // Lame 2
  digitalWrite(11, HIGH);
  digitalWrite(12, LOW);
  digitalWrite(13, HIGH);  // ON simple
}

void lamesOFF() {

  // Couper PWM (même si pas PWM)
  digitalWrite(2, LOW);
  digitalWrite(13, LOW);

  // Couper directions (TRÈS IMPORTANT)
  digitalWrite(8, LOW);
  digitalWrite(10, LOW);
  digitalWrite(11, LOW);
  digitalWrite(12, LOW);
}
